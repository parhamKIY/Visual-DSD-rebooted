cmake_minimum_required(VERSION 3.16)
project(visualdsd_sundials LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_C_FLAGS_RELEASE "/MT /O2 /DNDEBUG" CACHE STRING "" FORCE)
  set(CMAKE_CXX_FLAGS_RELEASE "/MT /O2 /DNDEBUG" CACHE STRING "" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
foreach(package ARKODE CVODES IDA IDAS KINSOL)
  set(BUILD_${package} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(language C CXX F77 F90)
  set(EXAMPLES_ENABLE_${language} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${SUNDIALS_SOURCE}" sundials EXCLUDE_FROM_ALL)
if(MSVC)
  set_property(TARGET sundials_cvode_static PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
add_library(visualdsd_sundials SHARED SundialsSolver.cpp)
target_compile_features(visualdsd_sundials PRIVATE cxx_std_14)
target_include_directories(visualdsd_sundials PRIVATE "${SUNDIALS_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}/sundials/include")
target_link_libraries(visualdsd_sundials PRIVATE sundials_cvode_static)
if(MSVC)
  target_compile_definitions(visualdsd_sundials PRIVATE SUNDIALSSOLVER15_EXPORTS)
  set_property(TARGET visualdsd_sundials PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
else()
  target_compile_definitions(visualdsd_sundials PRIVATE NIX)
  target_link_libraries(visualdsd_sundials PRIVATE m)
endif()
set_target_properties(visualdsd_sundials PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${NATIVE_OUTPUT}"
  LIBRARY_OUTPUT_DIRECTORY "${NATIVE_OUTPUT}")
